async function updateResourcesStock(event) { event.preventDefault(); const ipadVal = parseInt(document.getElementById('adminIpadStock').value); const mobileVal = parseInt(document.getElementById('adminMobileStock').value); const res1 = await _supabase.from('resources').update({ total_qty: ipadVal }).eq('name', 'iPad'); const res2 = await _supabase.from('resources').update({ total_qty: mobileVal }).eq('name', 'Mobile'); if (res1.error || res2.error) { alert('❌ 儲存失敗：' + (res1.error?.message || res2.error?.message)); } else { alert('✅ 全校物資總量已成功更新！前台已即時套用新上限。'); } }
